**Goal:** Build a locally-runnable web dashboard for LUMM mastermind group with mock data, ready for screen-sharing demo to colleagues.

function getThirdThursday(year: number, month: number): string {
  const d = new Date(year, month, 1);
  const firstDay = d.getDay();
  const firstThursday = firstDay <= 4 ? 4 - firstDay + 1 : 12 - firstDay;
  const thirdThursday = firstThursday + 14;
  return `${year}-${String(month + 1).padStart(2, "0")}-${String(thirdThursday).padStart(2, "0")}`;

import { db } from "@/db";
import { weeklyReports, members } from "@/db/schema";
import { eq, desc } from "drizzle-orm";
import { NextResponse } from "next/server";

export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);
  const memberId = searchParams.get("memberId");

  if (memberId) {
    const reports = db.select()
      .from(weeklyReports)
      .where(eq(weeklyReports.memberId, memberId))
      .orderBy(desc(weeklyReports.weekStart))
      .all();
    return NextResponse.json(reports);
  }

  // All reports with member info for group view
  const reports = db.select({
    report: weeklyReports,
    memberName: members.displayName,
    memberColor: members.avatarColor,
  })
    .from(weeklyReports)
    .innerJoin(members, eq(weeklyReports.memberId, members.id))
    .orderBy(desc(weeklyReports.weekStart))
    .all();

  return NextResponse.json(reports);
}

"use client";

type Props = {
  business: number;
  family: number;
  personal: number;
  size?: number;
};

export function MiniRadar({ business, family, personal, size = 120 }: Props) {
  const cx = size / 2;
  const cy = size / 2;
  const r = size * 0.38;

  // Triangle vertices (3 axes at 120 degrees apart, starting from top)
  const axes = [
    { x: cx, y: cy - r, label: "Б" },
    { x: cx + r * Math.sin((2 * Math.PI) / 3), y: cy + r * Math.cos((2 * Math.PI) / 3), label: "С" },
    { x: cx - r * Math.sin((2 * Math.PI) / 3), y: cy + r * Math.cos((2 * Math.PI) / 3), label: "Л" },
  ];

  const scores = [business / 10, family / 10, personal / 10];
  const dataPoints = axes.map((a, i) => ({
    x: cx + (a.x - cx) * scores[i],
    y: cy + (a.y - cy) * scores[i],
  }));

  const dataPath = dataPoints.map((p, i) => `${i === 0 ? "M" : "L"} ${p.x} ${p.y}`).join(" ") + " Z";

  // Grid lines at 33%, 66%, 100%
  const gridLevels = [0.33, 0.66, 1];

  return (
    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
      {/* Grid */}
      {gridLevels.map((level) => {
        const points = axes
          .map((a) => `${cx + (a.x - cx) * level},${cy + (a.y - cy) * level}`)
          .join(" ");
        return (
          <polygon
            key={level}
            points={points}
            fill="none"
            stroke="#3a3a3a"
            strokeWidth="0.5"
          />
        );
      })}

      {/* Axes */}
      {axes.map((a, i) => (
        <line key={i} x1={cx} y1={cy} x2={a.x} y2={a.y} stroke="#3a3a3a" strokeWidth="0.5" />
      ))}

      {/* Data area */}
      <path d={dataPath} fill="rgba(201, 168, 76, 0.2)" stroke="#c9a84c" strokeWidth="2" />

      {/* Data points */}
      {dataPoints.map((p, i) => (
        <circle key={i} cx={p.x} cy={p.y} r="3" fill="#c9a84c" />
      ))}

      {/* Labels */}
      {axes.map((a, i) => {
        const offset = 14;
        const lx = cx + (a.x - cx) * 1.25;
        const ly = cy + (a.y - cy) * 1.25;
        return (
          <text
            key={i}
            x={lx}
            y={ly}
            textAnchor="middle"
            dominantBaseline="middle"
            fill="#999"
            fontSize="11"
            fontWeight="500"
          >
            {a.label}:{[business, family, personal][i]}
          </text>
        );
      })}
    </svg>
  );
}

"use client";

type Props = {
  data: number[];
  width?: number;
  height?: number;
  color?: string;
};

export function Sparkline({ data, width = 150, height = 40, color = "#c9a84c" }: Props) {
  if (data.length < 2) return null;

  const min = Math.min(...data);
  const max = Math.max(...data);
  const range = max - min || 1;
  const padding = 4;

  const points = data.map((v, i) => {
    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
    const y = height - padding - ((v - min) / range) * (height - padding * 2);
    return `${x},${y}`;
  });

  const pathD = points.map((p, i) => `${i === 0 ? "M" : "L"} ${p}`).join(" ");

  // Gradient area
  const areaD = `${pathD} L ${width - padding},${height - padding} L ${padding},${height - padding} Z`;

  return (
    <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`}>
      <defs>
        <linearGradient id={`grad-${color.replace("#", "")}`} x1="0" y1="0" x2="0" y2="1">
          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
          <stop offset="100%" stopColor={color} stopOpacity="0" />
        </linearGradient>
      </defs>
      <path d={areaD} fill={`url(#grad-${color.replace("#", "")})`} />
      <path d={pathD} fill="none" stroke={color} strokeWidth="2" strokeLinecap="round" />
      {/* Last point dot */}
      <circle
        cx={parseFloat(points[points.length - 1].split(",")[0])}
        cy={parseFloat(points[points.length - 1].split(",")[1])}
        r="3"
        fill={color}
      />
    </svg>
  );
}

import { db } from "@/db";
import { meetings, members } from "@/db/schema";
import { eq, desc } from "drizzle-orm";

export const dynamic = "force-dynamic";

function daysUntil(dateStr: string): number {
  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
}

export default async function CalendarPage() {
  const allMeetings = db.select({
    meeting: meetings,
    organizerName: members.displayName,
    organizerColor: members.avatarColor,
  })
    .from(meetings)
    .leftJoin(members, eq(meetings.organizerId, members.id))
    .orderBy(desc(meetings.date))

                      month: "long",
                      year: "numeric",
